            if ($s === '' || !preg_match('/^[IVXLCDM]+$/', $s)) {
                throw new \InvalidArgumentException("Invalid characters or empty string");
            }

            $map = [
                'I' => 1, 'V' => 5, 'X' => 10, 'L' => 50,
                'C' => 100, 'D' => 500, 'M' => 1000
            ];

            $total = 0;
            $prevValue = 0;
            // Standard parsing logic
            // Iterate backwards or forwards?
            // Let's go backwards for ease of subtractive logic
            $chars = str_split($s);
            for ($i = count($chars) - 1; $i >= 0; $i--) {
                $currentValue = $map[$chars[$i]];
                if ($currentValue < $prevValue) {
                    $total -= $currentValue;
                } else {
                    $total += $currentValue;
                }
                $prevValue = $currentValue;
            }

            // Crucial check for canonicity:
            // If toRoman(total) doesn't match the input string, it's non-canonical
            if ($total < 1 || $total > 3999 || self::toRoman($total) !== $s) {
                throw new \InvalidArgumentException("Non-canonical Roman numeral");
            }

            return $total;
        }
