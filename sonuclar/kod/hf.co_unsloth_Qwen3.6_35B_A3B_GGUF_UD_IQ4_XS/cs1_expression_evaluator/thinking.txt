     public static class ExpressionEvaluator
     {
         private static readonly char[] Whitespace = { ' ', '\t', '\n', '\r' };
         private string _expression;
         private int _pos;

         public static double Evaluate(string expression)
         {
             if (string.IsNullOrWhiteSpace(expression))
                 throw new FormatException("Expression cannot be empty or whitespace.");

             var parser = new Parser(expression);
             return parser.ParseExpression();
         }

         private class Parser
         {
             private readonly string _expr;
             private int _pos;

             public Parser(string expr) { _expr = expr; _pos = 0; }

             public double ParseExpression() => ParseTerm();

             public double ParseTerm()
             {
                 double left = ParseUnary();
                 while (Peek() is '*' or '/')
                 {
                     char op = Consume();
                     double right = ParseUnary();
                     left = op == '*' ? left * right : left / right;
                 }
                 return left;
             }

             public double ParseUnary()
             {
                 if (Peek() is '+' or '-')
                 {
                     char op = Consume();
                     double val = ParseUnary();
                     return op == '-' ? -val : val;
                 }
                 return ParsePower();
             }

             public double ParsePower()
             {
                 double baseVal = ParseAtom();
                 if (Peek() == '^')
                 {
                     Consume();
                     double exp = ParsePower();
                     return Math.Pow(baseVal, exp);
                 }
                 return baseVal;
             }

             public double ParseAtom()
             {
                 SkipWhitespace();
                 if (Peek() == '(')
                 {
                     Consume(); // '('
                     double val = ParseExpression();
                     SkipWhitespace();
                     if (Peek() != ')')
                         throw new FormatException("Mismatched parentheses.");
                     Consume(); // ')'
                     return val;
                 }
                 return ParseNumber();
             }

             public double ParseNumber()
             {
                 SkipWhitespace();
                 int start = _pos;
                 bool hasDot = false;
                 bool hasDigit = false;

                 while (_pos < _expr.Length)
                 {
                     char c = _expr[_pos];
                     if (char.IsDigit(c))
                     {
                         hasDigit = true;
                         _pos++;
                     }
                     else if (c == '.' && !hasDot)
                     {
                         hasDot = true;
                         _pos++;
                     }
                     else
                     {
                         break;
                     }
                 }

                 if (!hasDigit)
                     throw new FormatException("Invalid number format.");

                 // Check for malformed like 1.2.3 (already handled by !hasDot)
                 // Check for trailing dot? e.g., "5." is usually valid, but let's be safe.
                 // The prompt says "malformed numbers (1.2.3)". It doesn't forbid 5.
                 // I'll allow 5. and .5.

                 string numStr = _expr.Substring(start, _pos - start);
                 if (double.TryParse(numStr, NumberStyles.Float, CultureInfo.InvariantCulture, out double result))
                     return result;
                 throw new FormatException("Invalid number format.");
             }

             private char Peek() => _pos < _expr.Length ? _expr[_pos] : '\0';
             private char Consume() => _expr[_pos++];
             private void SkipWhitespace() { while (_pos < _expr.Length && char.IsWhiteSpace(_expr[_pos])) _pos++; }
         }
     }

   using System;
   using System.Globalization;

   namespace Bench
   {
       public static class ExpressionEvaluator
       {
           public static double Evaluate(string expression)
           {
               if (string.IsNullOrWhiteSpace(expression))
                   throw new FormatException("Expression cannot be empty or whitespace.");

               var parser = new Parser(expression);
               double result = parser.ParseExpression();
               parser.SkipWhitespace();
               if (parser._pos < parser._expr.Length)
                   throw new FormatException("Unexpected characters at the end of the expression.");
               return result;
           }

           private class Parser
           {
               private readonly string _expr;
               private int _pos;

               public Parser(string expr) { _expr = expr; _pos = 0; }

               public double ParseExpression() => ParseTerm();

               public double ParseTerm()
               {
                   double left = ParseUnary();
                   while (Peek() is '*' or '/')
                   {
                       char op = Consume();
                       double right = ParseUnary();
                       if (op == '*') left *= right;
                       else
                       {
                           if (right == 0) throw new DivideByZeroException();
                           left /= right;
                       }
                   }
                   return left;
               }

               public double ParseUnary()
               {
                   if (Peek() is '+' or '-')
                   {
                       char op = Consume();
                       double val = ParseUnary();
                       return op == '-' ? -val : val;
                   }
                   return ParsePower();
               }

               public double ParsePower()
               {
                   double baseVal = ParseAtom();
                   if (Peek() == '^')
                   {
                       Consume();
                       double exp = ParsePower();
                       return Math.Pow(baseVal, exp);
                   }
                   return baseVal;
               }

               public double ParseAtom()
               {
                   SkipWhitespace();
                   if (Peek() == '(')
                   {
                       Consume();
                       double val = ParseExpression();
                       SkipWhitespace();
                       if (Peek() != ')')
                           throw new FormatException("Mismatched parentheses.");
                       Consume();
                       return val;
                   }
                   return ParseNumber();
               }

               public double ParseNumber()
               {
                   SkipWhitespace();
                   int start = _pos;
                   bool hasDot = false;
                   bool hasDigit = false;

                   while (_pos < _expr.Length)
                   {
                       char c = _expr[_pos];
                       if (char.IsDigit(c))
                       {
                           hasDigit = true;
                           _pos++;
                       }
                       else if (c == '.' && !hasDot)
                       {
                           hasDot = true;
                           _pos++;
                       }
                       else
                       {
                           break;
                       }
                   }

                   if (!hasDigit)
                       throw new FormatException("Invalid number format.");

                   // Check for malformed like 1.2.3
                   if (hasDot && _pos < _expr.Length && _expr[_pos] == '.')
                       throw new FormatException("Invalid number format.");

                   string numStr = _expr.Substring(start, _pos - start);
                   if (double.TryParse(numStr, NumberStyles.Float, CultureInfo.InvariantCulture, out double result))

                   throw new FormatException("Invalid number format.");
               }

               private char Peek() => _pos < _expr.Length ? _expr[_pos] : '\0';
               private char Consume() => _expr[_pos++];
               private void SkipWhitespace()

               }

               {

                   {
                       char op = Consume();
                       double right = ParseTerm();
                       left = op == '+' ? left + right : left - right;
